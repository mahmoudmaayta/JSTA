API (JSON; return {ok:true,data} / {ok:false,error})
POST /api/forms/office-2026          // upsert office + branches + create consent DATA_ACCURACY
POST /api/forms/staff-2026           // upsert people + roles (4 segments; 12 columns each row)
POST /api/forms/commitment           // create consent COMPLAINT_COMMITMENT with complaint numbers
